{{/* Helper to format experience from numeric value to localized text Usage: {{ partial "helpers/format-experience.html" (dict "experience" .experience "context" $) }} Or simple usage (defaults to French): {{ partial "helpers/format-experience.html" .experience }} Input: Numeric experience value (e.g., 2.5) or dict with experience and context Output: Formatted localized text (e.g., "2 years and 6 months" or "2 ans et 6 mois") */}} {{- $experience := . -}} {{- $context := . -}} {{/* Handle both simple call (number) and dict call (with context) */}} {{- if reflect.IsMap . -}} {{- $experience = .experience -}} {{- $context = .context | default . -}} {{- end -}} {{- $experienceText := "" -}} {{/* Always use i18n for localization - Hugo automatically uses current language context */}} {{- $yearSingular := i18n "year_singular" | default "an" -}} {{- $yearsPlural := i18n "years_plural" | default "ans" -}} {{- $monthSingular := i18n "month_singular" | default "mois" -}} {{- $monthsPlural := i18n "months_plural" | default "mois" -}} {{- $andConnector := i18n "and_connector" | default "et" -}} {{- if $experience -}} {{- $exp := float $experience -}} {{- $years := int (math.Floor $exp) -}} {{- $months := int (math.Round (mul (sub $exp (math.Floor $exp)) 12)) -}} {{- if eq $years 0 -}} {{- if eq $months 1 -}} {{- $experienceText = printf "1 %s" $monthSingular -}} {{- else -}} {{- $experienceText = printf "%d %s" $months $monthsPlural -}} {{- end -}} {{- else if eq $months 0 -}} {{- if eq $years 1 -}} {{- $experienceText = printf "1 %s" $yearSingular -}} {{- else -}} {{- $experienceText = printf "%d %s" $years $yearsPlural -}} {{- end -}} {{- else -}} {{- if eq $years 1 -}} {{- if eq $months 1 -}} {{- $experienceText = printf "1 %s %s 1 %s" $yearSingular $andConnector $monthSingular -}} {{- else -}} {{- $experienceText = printf "1 %s %s %d %s" $yearSingular $andConnector $months $monthsPlural -}} {{- end -}} {{- else -}} {{- if eq $months 1 -}} {{- $experienceText = printf "%d %s %s 1 %s" $years $yearsPlural $andConnector $monthSingular -}} {{- else -}} {{- $experienceText = printf "%d %s %s %d %s" $years $yearsPlural $andConnector $months $monthsPlural -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $experienceText -}}
