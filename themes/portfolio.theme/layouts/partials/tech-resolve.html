{{- /* Résout une clé techno (lang_* / fw_* / spec_* / skill_* / tool_*) → métadonnées. Contexte = la clé (string). Retourne dict { key, label, iconPath, iconSrc, icon, color, experience }. iconPath = chemin brut déclaré dans data/*.json (sert au test de présence). iconSrc = URL à mettre dans src : icône redimensionnée par le pipeline Hugo. Les icônes sont rendues entre 14 et 22 px ; les servir en pleine résolution faisait décoder des bitmaps énormes (Blazor.png : 1000x1000, soit ~3,8 Mo en mémoire pour 16 px à l'écran). get-optimized-image laisse passer les SVG. */ -}} {{- $key := . -}} {{- $found := dict -}} {{- $sources := slice site.Data.programming_languages site.Data.frameworks_engines site.Data.specialties site.Data.soft_skills site.Data.tools -}} {{- range $sources -}} {{- range . -}} {{- if eq .key $key -}}{{- $found = . -}}{{- end -}} {{- end -}} {{- end -}} {{- $iconPath := $found.iconPath -}} {{- $iconSrc := "" -}} {{- if $iconPath -}} {{- if not (hasPrefix $iconPath "/") -}}{{- $iconPath = printf "/%s" $iconPath -}}{{- end -}} {{- /* 72 px couvre le plus grand rendu (22 px) sur un écran à 3x. */ -}} {{- $iconSrc = (partial "get-optimized-image.html" (dict "imagePath" $iconPath "width" 72 "quality" 85)).src -}} {{- end -}} {{- $label := i18n $key -}} {{- if not $label -}}{{- $label = $key | replaceRE "^(lang_|fw_|spec_|skill_|tool_)" "" | humanize | title -}}{{- end -}} {{- return dict "key" $key "label" $label "iconPath" $iconPath "iconSrc" $iconSrc "icon" $found.icon "color" $found.color "experience" $found.experience -}}
